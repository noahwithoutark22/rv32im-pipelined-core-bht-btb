.text
.globl _start

_start:

####################################################
# Base addresses
####################################################
addi s0, x0, 0          # A base = 0
addi s1, x0, 36         # B base = 9*4 = 36
addi s2, x0, 72         # C base = 18*4 = 72

####################################################
# Store Matrix A into Data Memory
####################################################

addi t0,x0,1
sw t0,0(s0)

addi t0,x0,2
sw t0,4(s0)

addi t0,x0,3
sw t0,8(s0)

addi t0,x0,4
sw t0,12(s0)

addi t0,x0,5
sw t0,16(s0)

addi t0,x0,6
sw t0,20(s0)

addi t0,x0,7
sw t0,24(s0)

addi t0,x0,8
sw t0,28(s0)

addi t0,x0,9
sw t0,32(s0)

####################################################
# Store Matrix B into Data Memory
####################################################

addi t0,x0,9
sw t0,0(s1)

addi t0,x0,8
sw t0,4(s1)

addi t0,x0,7
sw t0,8(s1)

addi t0,x0,6
sw t0,12(s1)

addi t0,x0,5
sw t0,16(s1)

addi t0,x0,4
sw t0,20(s1)

addi t0,x0,3
sw t0,24(s1)

addi t0,x0,2
sw t0,28(s1)

addi t0,x0,1
sw t0,32(s1)

####################################################
# Matrix Multiplication
####################################################

addi t0,x0,0            # i = 0

LOOP_I:

addi t1,x0,0            # j = 0

LOOP_J:

addi t2,x0,0            # sum = 0

addi t3,x0,0            # k = 0

LOOP_K:

####################################################
# Address of A[i][k]
####################################################

addi t4,x0,3
mul t5,t0,t4            # i*3
add t5,t5,t3            # i*3+k
slli t5,t5,2            # *4
add t6,s0,t5
lw t6,0(t6)

####################################################
# Address of B[k][j]
####################################################

addi t4,x0,3
mul t5,t3,t4            # k*3
add t5,t5,t1            # k*3+j
slli t5,t5,2
add a0,s1,t5
lw a0,0(a0)

####################################################
# sum += A*B
####################################################

mul a1,t6,a0
add t2,t2,a1

####################################################
# k++
####################################################

addi t3,t3,1

addi t4,x0,3
blt t3,t4,LOOP_K

####################################################
# Store C[i][j]
####################################################

addi t4,x0,3
mul t5,t0,t4
add t5,t5,t1
slli t5,t5,2
add t6,s2,t5

sw t2,0(t6)

####################################################
# j++
####################################################

addi t1,t1,1

addi t4,x0,3
blt t1,t4,LOOP_J

####################################################
# i++
####################################################

addi t0,t0,1

addi t4,x0,3
blt t0,t4,LOOP_I

####################################################
# End
####################################################

END:
beq x0,x0,END
